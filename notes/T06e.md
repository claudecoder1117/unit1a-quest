**Solutions (`solution[] = {say, math}`)** — 7 steps per factoring card: GCF check → a · c → the pair with its arithmetic shown → split the middle term → factor by grouping → pull out the common binomial (this step's `math` **is the teacher's key string**) → FOIL check ending on the original trinomial with ✓. const bgcd = (a, b) => { a = a < 0n ? -a : a; b = b < 0n ? -b : b; while (b) [a, b] = [b, a % b]; return a; };
const Q = (n, d = 1n) => { n = BigInt(n); d = BigInt(d); if (d < 0n) { n = -n; d = -d; } const g = bgcd(n, d) || 1n; return { n: n / g, d: d / g }; };

// ---------- normalize + tokenize + parse (S3 grammar) ----------
function normalize(s) {
  // superscript glyphs BEFORE NFKC (NFKC maps ² → 2, which would turn 3p² into 3p2)
  return s.replace(/²/g, '^2').replace(/³/g, '^3').normalize('NFKC').replace(/[−–—]/g, '-').replace(/[·×]/g, '*')
    .replace(/\[/g, '(').replace(/\]/g, ')').replace(/\s+/g, '');
}
function tokenize(s) {
  const t = []; let i = 0;
  while (i < s.length) {
    const ch = s[i];
    if (/\d/.test(ch)) { let j = i; while (j < s.length && /[\d.]/.test(s[j])) j++; t.push({ k: 'NUM', v: s.slice(i, j) }); i = j; continue; }
    if (/[a-zA-Z]/.test(ch)) { t.push({ k: 'VAR', v: ch }); i++; continue; }
    if ('+-*/^()'.includes(ch)) { t.push({ k: ch }); i++; continue; }
    throw new Error(`bad char ${JSON.stringify(ch)} in ${s}`);
  }
  return t;
}
function parse(src) {
  const t = tokenize(normalize(src)); let i = 0; const vars = new Set();
  const peek = () => t[i]; const next = () => t[i++];
  const startsAtom = (tok) => tok && (tok.k === 'NUM' || tok.k === 'VAR' || tok.k === '(');

  function term() { let a = unary(); for (;;) { if (peek() && peek().k === '*') { next(); a = P.mul(a, unary()); } else if (startsAtom(peek()) && peek().k !== 'NUM') { a = P.mul(a, unary()); } else if (peek() && peek().k === 'NUM' && t[i - 1] && t[i - 1].k === ')') { a = P.mul(a, unary()); } else break; } return a; }
  function unary() { if (peek() && peek().k === '-') { next(); return P.neg(unary()); } if (peek() && peek().k === '+') { next(); return unary(); } return pow(); }
  function pow() { const a = atom(); if (peek() && peek().k === '^') { next(); const e = next(); assert.equal(e.k, 'NUM'); return P.pow(a, Number(e.v)); } return a; }
  function atom() {

    throw new Error(`unexpected token ${tok.k} in ${src}`);
  }
  const numQ = (v) => { if (v.includes('.')) { const [a, b] = v.split('.'); return Q(BigInt(a + b), 10n ** BigInt(b.length)); } return Q(BigInt(v)); };
  const poly = expr(); assert.equal(i, t.length, `trailing tokens in ${src}`);
  return { poly, vars: [...vars] };
}
const expand = (s) => parse(s).poly;

// factor structure of a key: leading sign, integer constant, then linear factors (p v + q) with gcd(p,q)=1
function factorStructure(s, v) {
  const n = normalize(s);
  const m = n.match(new RegExp(`^(-?)(\\d*)((?:\\(-?\\d*${v}[+-]\\d+\\))+)$`));
  assert(m, `key is not a product of linear factors: ${s}`);
  const factors = [...m[3].matchAll(new RegExp(`\\((-?\\d*)${v}([+-])(\\d+)\\)`, 'g'))].map((f) => {

  });
  return { neg: m[1] === '-', g: m[2] === '' ? 1 : Number(m[2]), factors };
}

import { test } from 'node:test';
import { existsSync } from 'node:fs';
import { fileURLToPath, pathToFileURL } from 'node:url';

const SITE_CANDIDATES = [
  process.env.SITE_DIR,
  fileURLToPath(new URL('../site/', import.meta.url)),
  '/Users/oliver/Projects/unit1a-quest/site/',

  = await import(pathToFileURL(SITE + 'js/grader/normalize.js').href);

// polynomial = array of Rat by power
const trim = (a) => { while (a.length > 1 && a[a.length - 1].n === 0) a.pop(); return a; };
const POLY = {
  num: (t) => [evalNum(t)],
  variable: () => [rat(0), rat(1)],
  neg: (a) => a.map(ratNeg),
  add: (a, b) => trim(Array.from({ length: Math.max(a.length, b.length) }, (_, i) => ratAdd(a[i] ?? rat(0), b[i] ?? rat(0)))),
  sub: (a, b) => POLY.add(a, POLY.neg(b)),
  mul: (a, b) => { const r = Array.from({ length: a.length + b.length - 1 }, () => rat(0)); a.forEach((x, i) => b.forEach((y, j) => { r[i + j] = ratAdd(r[i + j], ratMul(x, y)); })); return trim(r); },
  div: (a, b) => { assert.equal(b.length, 1, 'division by a constant only'); return a.map((x) => ratDiv(x, b[0])); },
  pow: (a, k) => { let r = [rat(1)]; for (let i = 0; i < k; i++) r = POLY.mul(r, a); return r; },
};
const evalNum = (t) => { const [i, f = ''] = t.split('.'); return rat(Number(i + f), 10 ** f.length); };
const peq = (a, b) => a.length === b.length && a.every((x, i) => ratEq(x, b[i]));
const pstr = (a, v) => a.map((q, i) => `${ratToString(q)}${i ? v : ''}${i > 1 ? '^' + i : ''}`).reverse().join(' + ');

function expand(text, expectVar) {
  const norm = normalizeText(text, { prefix: false, units: false, degrees: false });   // the grader-side pipeline minus the num-only rules
  const ast = parseExpr(norm);
  const vars = astVars(ast);
  assert.deepEqual(vars, expectVar ? [expectVar] : [], `${text} → variables ${vars}`);
  return { poly: evalAst(ast, POLY), norm };
}

  }
